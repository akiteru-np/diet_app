        /* 🧹 削除機能用のリストスタイル */

<body>


    <div class="container">

        <h2>📊 体重ダッシュボード</h2>

        

        <div class="form-group">

            <input type="date" id="dateInput">

            <input type="number" id="weightInput" step="0.1" placeholder="体重 (kg)">

            <button onclick="saveWeight()">登録</button>

        </div>


        <label style="font-size: 13px; font-weight: bold; color: #7f8c8d;">🗑️ 履歴の管理（クリックで消去）</label>

        <ul id="weightDataList" class="data-list"></ul>


        <canvas id="weightChart"></canvas>

    </div>


    <script>

        let weightChart; // グラフのデータを入れおく箱


        // ① ページが開かれたら、裏側からデータを取ってきてグラフと管理リストを描く命令

        async function fetchAndRenderChart() {

            // Render上の本番API（またはローカルAPI）にアクセスしてデータをもらう

            // ※index.htmlを直接開く場合、URLのドメインをRenderのURLにする必要があります

            const baseUrl = window.location.hostname === "127.0.0.1" || window.location.hostname === "localhost" ? "" : "https://diet-app-pfc.onrender.com";

            

            const response = await fetch(`${baseUrl}/weights/`);

            const data = await response.json();


            // 日付順に並び替え

            data.sort((a, b) => new Date(a.date) - new Date(b.date));


            // 🗂️ 画面上のミニ管理リストを更新

            const listContainer = document.getElementById('weightDataList');

            listContainer.innerHTML = '';

            data.forEach(item => {

                const li = document.createElement('li');

                li.className = 'data-item';

                li.innerHTML = `

                    <span>📅 ${item.date} : <strong>${item.weight} kg</strong></span>

                    <button class="delete-btn" onclick="deleteWeight(${item.id})">🗑️ 削除</button>

                `;

                listContainer.appendChild(li);

            });


            // グラフ用の「日付リスト」と「体重リスト」を作る

            const labels = data.map(item => item.date);

            const weights = data.map(item => item.weight);


            // グラフを描く画用紙の準備

            const ctx = document.getElementById('weightChart').getContext('2d');

            

            // 古いグラフがあったら一旦消す（更新するため）

            if (weightChart) {

                weightChart.destroy();

            }


            // 新しい折れ線グラフを描く！

            weightChart = new Chart(ctx, {

                type: 'line',

                data: {

                    labels: labels,

                    datasets: [{

                        label: '体重 (kg)',

                        data: weights,

                        borderColor: '#27ae60',       // 線の色（緑）

                        backgroundColor: 'rgba(39, 174, 96, 0.2)', // 線のしたの塗りつぶし色

                        borderWidth: 2,

                        tension: 0.1 // 少し曲線をなめらかにする

                    }]

                },

                options: {

                    scales: {

                        y: { beginAtZero: false } // 体重なので0kgから始めない設定

                    }

                }

            });

        }


        // ② 「登録」ボタンが押された時の命令

        async function saveWeight() {

            const dateInput = document.getElementById('dateInput').value;

            const weightInput = document.getElementById('weightInput').value;


            if (!dateInput || !weightInput) {

                alert("日付と体重を両方入力してください！");

                return;

            }


            const baseUrl = window.location.hostname === "127.0.0.1" || window.location.hostname === "localhost" ? "" : "https://diet-app-pfc.onrender.com";


            const response = await fetch(`${baseUrl}/weights/`, {

                method: 'POST',

                headers: { 'Content-Type': 'application/json' },

                body: JSON.stringify({

                    date: dateInput,

                    weight: parseFloat(weightInput)

                })

            });


            if (response.ok) {

                alert("登録しました！");

                fetchAndRenderChart(); 

            } else {

                alert("エラーが発生しました。");

            }

        }


        // 🗑️ ③ 【新機能】ゴミ箱ボタンを押した時に裏側へ削除命令を飛ばす魔法

        async function deleteWeight(id) {

            if (!confirm("本当にこの体重データを削除しますか？")) return;

            

            const baseUrl = window.location.hostname === "127.0.0.1" || window.location.hostname === "localhost" ? "" : "https://diet-app-pfc.onrender.com";


            const response = await fetch(`${baseUrl}/weights/${id}`, {

                method: 'DELETE'

            });


            if (response.ok) {

                alert("データベースから削除しました！");

                fetchAndRenderChart(); // グラフとリストを再描画

            } else {

                alert("削除に失敗しました。");

            }

        }


        // ④ 画面が開いた瞬間に自動で実行

        window.onload = fetchAndRenderChart;

    </script>
